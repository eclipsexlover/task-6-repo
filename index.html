let input = document.getElementById('textInput'); let button = document.getElementById('addButton'); button.style.display = 'none'; input.oninput = function() { if (input.value === '') { button.style.display = 'none'; } else { button.style.display = 'block'; } }; button.onclick = function() { let newParagraph = document.createElement('p'); newParagraph.textContent = input.value; document.body.insertBefore(newParagraph, input); input.value = ''; button.style.display = 'none'; };
